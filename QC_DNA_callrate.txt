############################################################
# 4. Clean and define variables
############################################################

# Convert LABEL_ID to character and remove leading/trailing spaces.

data$call_rate_raw <- as.numeric(data$call_rate_raw)

# Convert QC_pass to logical if needed and create QC_fail.
# QC_fail is the opposite of QC_pass.

  mutate(
    LABEL_ID = as.character(LABEL_ID),
    totdna_ng = as.numeric(totdna_ng),
    call_rate_raw = as.numeric(call_rate_raw),
    QC_pass = as.logical(QC_pass),
    QC_fail = !QC_pass

# Count QC pass/fail after cleaning.
table(df$QC_pass)


############################################################
# 5. Explore call rate by QC status
############################################################

# Summarise call rate statistics separately for QC pass and QC fail samples.
# This helps evaluate whether failed samples generally have lower call rates.
df %>%
  group_by(QC_pass) %>%

    min_call = min(call_rate_raw, na.rm = TRUE),

# Count how many samples have a call rate below the common threshold of 0.98.
# This threshold was part of the QC criteria.

# Interpretation:
# If the number of QC failures differs from the number of samples below
# call rate 0.98, QC status is not determined exclusively by call rate.
# Other QC metrics may also contribute, such as heterozygosity or sex checks.

############################################################
# 16. Threshold analysis: DNA amount vs QC failure
############################################################

# The purpose of this analysis is to evaluate what would happen if
# minimum DNA amount thresholds were applied before genotyping.
# For each threshold, the code calculates:
#   - number of retained samples
#   - fraction of retained samples
#   - QC failure rate among retained samples
#   - number of QC failures avoided by dropping low-DNA samples
#   - number of QC-pass samples lost by dropping low-DNA samples

thresholds <- c(50, 100, 150, 200)

results <- lapply(thresholds, function(t) {

  # Samples retained if their DNA amount is equal to or above threshold.
  kept <- df[df$totdna_ng >= t, ]

  # Samples dropped if their DNA amount is below threshold.
  dropped <- df[df$totdna_ng < t, ]

  data.frame(
    threshold_ng = t,
    n_kept = nrow(kept),
    kept_fraction = nrow(kept) / nrow(df),
    fail_rate_kept = mean(kept$QC_fail, na.rm = TRUE),
    fails_avoided = sum(dropped$QC_fail, na.rm = TRUE),
    passes_lost = sum(dropped$QC_pass, na.rm = TRUE)
  )
})

# Combine the list of results into one data frame.
results <- do.call(rbind, results)

# View threshold analysis results.
results

# Export threshold analysis to CSV.
write.csv(results, "threshold_analysis.csv", row.names = FALSE)
