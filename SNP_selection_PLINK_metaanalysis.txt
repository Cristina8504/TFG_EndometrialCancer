# This block reads a previous PLINK association file.
# It is kept for traceability, but the final analysis below uses PLINK2 GLM.
assoc_new <- read.table(

# Keep only additive model results.

############################################################
# 7. Read PLINK2 GLM logistic results from Round 1
############################################################

glm1 <- read.table(
  round1_glm_file,
  header = TRUE,
  stringsAsFactors = FALSE,
  comment.char = ""
)

# Rename first column if PLINK uses "#CHROM".
names(glm1)[1] <- "CHROM"

# Keep only additive model.
glm_add <- subset(glm1, TEST == "ADD")

# Clean SNP IDs by removing technical prefixes.
glm_add$ID <- gsub("^GSA-|^exm-", "", glm_add$ID)

# Convert beta to odds ratio.
glm_add$OR <- exp(glm_add$BETA)

# Compact table for Round 1.
tabla_glm_round1 <- glm_add[, c("ID", "A1", "A1_FREQ", "OR", "P")]


############################################################
# 8. Read PLINK2 GLM logistic results from Round 2
############################################################

glm2 <- read.table(
  round2_glm_file,
  header = TRUE,
  stringsAsFactors = FALSE,
  comment.char = ""
)

names(glm2)[1] <- "CHROM"

# Keep only additive model.
glm2_add <- subset(glm2, TEST == "ADD")

# Clean SNP IDs and calculate OR.
glm2_add$ID <- gsub("^GSA-|^exm-", "", glm2_add$ID)
glm2_add$OR <- exp(glm2_add$BETA)

# Compact table for Round 2.
tabla_glm_round2 <- glm2_add[, c("ID", "A1", "A1_FREQ", "OR", "P")]

  rowNames = FALSE
)


############################################################
# 9. Prepare Round 1 and Round 2 tables for meta-analysis
############################################################

# Keep only the variables needed for inverse-variance meta-analysis.
r1 <- glm_add[, c("ID", "A1", "BETA", "SE", "P")]
r2 <- glm2_add[, c("ID", "A1", "BETA", "SE", "P")]

# Clean SNP IDs again for safety.
r1$ID <- gsub("^GSA-|^exm-", "", r1$ID)
r2$ID <- gsub("^GSA-|^exm-", "", r2$ID)


############################################################
# 10. Merge Round 1 and Round 2 association results
############################################################

# Merge by SNP ID.
meta_df <- merge(r1, r2, by = "ID", suffixes = c("_R1", "_R2"))

# Check if effect alleles are the same between rounds.
meta_df$Same_A1 <- meta_df$A1_R1 == meta_df$A1_R2
meta_df[, c("ID", "A1_R1", "A1_R2", "Same_A1")]

# Important:
# Betas can only be combined directly if they refer to the same effect allele.
# If Same_A1 is FALSE, allele harmonization is required.
if (any(meta_df$Same_A1 == FALSE)) {
  warning("Some SNPs have different effect alleles between rounds. Harmonization is required before combining betas.")
}

# In this script, Round 2 beta is assumed to be aligned.
# If harmonization is required, replace this column with harmonized betas.
meta_df$BETA_R2_aligned <- meta_df$BETA_R2


############################################################
# 11. Fixed-effect inverse-variance meta-analysis
############################################################

# Calculate inverse-variance weights.
meta_df$w1 <- 1 / (meta_df$SE_R1^2)
meta_df$w2 <- 1 / (meta_df$SE_R2^2)

# Weighted fixed-effect beta.
meta_df$BETA_meta <- (
  meta_df$w1 * meta_df$BETA_R1 +
    meta_df$w2 * meta_df$BETA_R2_aligned
) / (meta_df$w1 + meta_df$w2)

# Standard error of combined beta.
meta_df$SE_meta <- sqrt(1 / (meta_df$w1 + meta_df$w2))

# Odds ratio, Z statistic and p-value.
meta_df$OR_meta <- exp(meta_df$BETA_meta)
meta_df$Z_meta <- meta_df$BETA_meta / meta_df$SE_meta
meta_df$P_meta <- 2 * pnorm(-abs(meta_df$Z_meta))

# 95% confidence interval for OR.
meta_df$CI_lower <- exp(meta_df$BETA_meta - 1.96 * meta_df$SE_meta)
meta_df$CI_upper <- exp(meta_df$BETA_meta + 1.96 * meta_df$SE_meta)

# Bonferroni correction.
meta_df$P_meta_bonf <- p.adjust(meta_df$P_meta, method = "bonferroni")


############################################################
# 12. Direction consistency between rounds
############################################################

meta_df$Direction_R1 <- ifelse(meta_df$BETA_R1 > 0, "Risk", "Protective")
meta_df$Direction_R2 <- ifelse(meta_df$BETA_R2_aligned > 0, "Risk", "Protective")
meta_df$Direction_consistent <- meta_df$Direction_R1 == meta_df$Direction_R2


############################################################
# 13. Create final meta-analysis table
############################################################

meta_table <- meta_df[, c(
  "ID",
  "A1_R1", "A1_R2", "Same_A1",
  "BETA_R1", "SE_R1", "P_R1",
  "BETA_R2_aligned", "SE_R2", "P_R2",
  "OR_meta", "CI_lower", "CI_upper",
  "BETA_meta", "SE_meta", "P_meta", "P_meta_bonf",
  "Direction_R1", "Direction_R2", "Direction_consistent"
)]

# Order by meta-analysis p-value.
meta_table <- meta_table[order(meta_table$P_meta), ]

write.xlsx(
  meta_table,
  file = "meta_analysis_results_clean_from_plink2.xlsx",
  rowNames = FALSE
)
