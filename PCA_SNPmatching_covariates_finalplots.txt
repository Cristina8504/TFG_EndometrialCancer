# SNP categories used for colouring the forest plot.
snp_categories <- data.frame(
  ID = c(
    "rs1219651", "rs12682374", "rs657152", "rs1123573",
    "rs11651755", "rs78540526", "rs11927381", "rs7463708",

    "rs4430796", "rs72915912", "rs1679014",
    "rs117610694", "rs12184995",
    "rs7501939", "rs143384", "rs7612543", "rs4135275",
    "rs62466318", "rs34670419", "rs7183977", "rs12449442",
    "rs9349553"
  ),
  Category = c(
    rep("General cancer", 11),
    rep("Primary EC", 5),
    rep("Pleiotropic", 9)
  )
)

# Function to prepare forest plot data from a GLM table.
prepare_forest <- function(glm_object, round_name) {
  glm_object %>%
    filter(TEST == "ADD") %>%
    mutate(
      ID = gsub("^GSA-|^exm-", "", ID),
      OR = exp(BETA),
      CI_lower = exp(BETA - 1.96 * SE),
      CI_upper = exp(BETA + 1.96 * SE),
      Round = round_name
    ) %>%
    left_join(snp_categories, by = "ID") %>%
    select(ID, OR, CI_lower, CI_upper, P, Round, Category)
}

# Prepare Round 1 and Round 2 forest plot data.
if (exists("glm1") && exists("glm2")) {
  forest_r1 <- prepare_forest(glm1, "Round 1")
  forest_r2 <- prepare_forest(glm2, "Round 2")

  # Combine both rounds.
  forest_df <- bind_rows(forest_r1, forest_r2)

  # Order categories and SNPs.
  forest_df <- forest_df %>%
    mutate(
      Category = factor(
        Category,
        levels = c("General cancer", "Primary EC", "Pleiotropic")
      )
    ) %>%
    arrange(Category, OR) %>%
    mutate(ID = factor(ID, levels = unique(ID)))

  # Forest plot.
  forest_plot <- ggplot(forest_df, aes(x = OR, y = ID, color = Category)) +
    geom_vline(xintercept = 1, linetype = "dashed") +
    geom_errorbarh(aes(xmin = CI_lower, xmax = CI_upper), height = 0.2) +
    geom_point(size = 2.5) +
    facet_wrap(~Round) +
    scale_x_log10() +
    labs(
      title = "Odds ratios from association analysis",
      x = "Odds Ratio (log scale)",
      y = "SNP",
      color = "Category"
    ) +
    theme_minimal() +
    theme(
      plot.title = element_text(size = 14, face = "bold"),
      axis.text.y = element_text(size = 9),
      legend.position = "right"
    )

  forest_plot

  ggsave(
    "forest_plot_round1_round2.png",
    plot = forest_plot,
    width = 10,
    height = 7,
    dpi = 300
  )
}

# Load meta-analysis results.
# The expected sheet should contain columns:
#   SNP or ID
#   OR_meta
#   P_meta
#   95% CI
metaplot <- read_excel("meta_analysis_results2.xlsx", sheet = "Sheet6")

# Harmonise SNP column name if needed.
if ("ID" %in% names(metaplot) && !"SNP" %in% names(metaplot)) {
  metaplot <- metaplot %>% rename(SNP = ID)
}

# Prepare confidence intervals from text column if needed.
# Expected format: "(0.52–0.86)" or "(0.52-0.86)".
metaplot <- metaplot %>%
  mutate(
    OR_meta = as.numeric(OR_meta),
    P_meta = as.numeric(P_meta)
  )

if ("95% CI" %in% names(metaplot)) {
  metaplot <- metaplot %>%
    mutate(
      CI_clean = str_remove_all(`95% CI`, "\\(|\\)"),
      CI_clean = str_replace_all(CI_clean, "–", "-"),
      CI_lower = as.numeric(str_trim(str_split_fixed(CI_clean, "-", 2)[, 1])),
      CI_upper = as.numeric(str_trim(str_split_fixed(CI_clean, "-", 2)[, 2]))
    )
}

# Add direction variable and order SNPs by p-value.
metaplot <- metaplot %>%
  mutate(
    Direction = case_when(
      OR_meta < 1 ~ "Protective",
      OR_meta > 1 ~ "Risk",
      TRUE ~ "No association"
    )
  ) %>%
  arrange(P_meta) %>%
  mutate(
    SNP = factor(SNP, levels = rev(SNP))
  )

# Create meta-analysis forest plot.
meta_forest_plot <- ggplot(metaplot, aes(x = OR_meta, y = SNP, color = Direction)) +
  geom_errorbarh(
    aes(xmin = CI_lower, xmax = CI_upper),
    height = 0.22,
    linewidth = 0.8
  ) +
  geom_point(size = 3.2) +
  geom_vline(
    xintercept = 1,
    linetype = "dashed",
    color = "black",
    linewidth = 0.6
  ) +
  scale_x_log10(
    breaks = c(0.5, 0.7, 1, 1.5, 2, 3),
    limits = c(0.45, 4)
  ) +
  labs(
    title = "Meta-analysis results for candidate SNPs",
    x = "Odds ratio (log scale)",
    y = NULL,
    color = "Direction"
  ) +
  theme_minimal(base_size = 13) +
  theme(
    plot.title = element_text(face = "bold", hjust = 0.5, size = 16),
    axis.text.y = element_text(size = 9),
    panel.grid.minor = element_blank(),
    legend.position = "bottom",
    legend.title = element_text(face = "bold")
  )

meta_forest_plot
