############################################################
# 5. Create simplified cytology and HPV groups
############################################################

# Create broader cytology categories from SNOMED_SEVERITY.
# Interpretation used in this script:
#   < 4  -> poor_quality
#   = 4  -> normal
#   > 5  -> abnormal
#   other values -> other
#
# Create broad HPV groups using HPVDIAG text:
#   contains "pos" -> positive
#   contains "neg" -> negative
#   otherwise      -> other
df <- df %>%
  mutate(
    cytology_group = case_when(
      is.na(SNOMED_SEVERITY) ~ NA_character_,
      SNOMED_SEVERITY < 4 ~ "poor_quality",
      SNOMED_SEVERITY == 4 ~ "normal",
      SNOMED_SEVERITY > 5 ~ "abnormal",
      TRUE ~ "other"
    ),
    hpv_group = case_when(
      is.na(HPVDIAG) ~ NA_character_,
      grepl("pos", HPVDIAG, ignore.case = TRUE) ~ "positive",
      grepl("neg", HPVDIAG, ignore.case = TRUE) ~ "negative",
      TRUE ~ "other"
    )
  )

# Define order of factor levels.
df$cytology_group <- factor(
  df$cytology_group,
  levels = c("poor_quality", "normal", "abnormal", "other")
)

df$hpv_group <- factor(
  df$hpv_group,
  levels = c("negative", "positive", "other")
)

# Check distribution of the newly created groups.
table(df$cytology_group, useNA = "ifany")
table(df$hpv_group, useNA = "ifany")

############################################################
# 13. Logistic model including cohort/case variable
############################################################

# Logistic regression model with QC pass as outcome.
model_qc_cohort <- glm(
  qc_pass ~ cohort_binary + HPVDIAG + SNOMED_WORST +
    dna_total_ng + year + collector_type + AGE,
  data = data1,
  family = binomial
)

summary(model_qc_cohort)

# Calculate odds ratios and 95% confidence intervals.
odds_ratios <- exp(cbind(
  OR = coef(model_qc_cohort),
  confint(model_qc_cohort)
))

# Convert OR results to a table.
odds_table <- as.data.frame(odds_ratios)

# Add variable names.
odds_table$Variable <- rownames(odds_table)

# Rename and clean columns.
odds_table <- odds_table %>%
  rename(
    OR = OR,
    CI_lower = `2.5 %`,
    CI_upper = `97.5 %`
  ) %>%
  select(Variable, OR, CI_lower, CI_upper) %>%
  mutate(
    across(c(OR, CI_lower, CI_upper), round, digits = 3)
  )

# View odds ratio table.
odds_table

# Export odds ratio table.
write_xlsx(odds_table, "odds_ratios_table_qc_cohort.xlsx")


############################################################
# 14. Add cohort_binary to main dataframe

# Add cohort_binary from data1 into df.
# This assumes rows are in the same order in df and data1.

# Create qc_fail variable:
#   1 = QC fail
#   0 = QC pass
df$qc_fail <- ifelse(df$qc_pass == FALSE, 1, 0)


############################################################
# 15. Create refined HPV variable from translation
############################################################

# Create hpv_final using the detailed translation variable.
df <- df %>%
  mutate(
    hpv_final = case_when(
      is.na(translation) ~ "Missing",
      translation == "HPV high risk types-negative" ~ "Negative",
      translation == "HPV high risk types-positive" ~ "Other_high_risk",
      translation == "HPV high risk type other than HPV16 or 18-positive" ~ "Other_high_risk",
      translation == "HPV 16-positive" ~ "HPV16",
      translation == "HPV 18-positive" ~ "HPV18",
      TRUE ~ "Other"
    )
  )

# HPV table.
tabla_hpv <- as.data.frame(table(df$hpv_final))
tabla_hpv


############################################################
# 16. Log-transform DNA total
############################################################

# DNA amount is usually right-skewed.
# log(x + 1) is used to reduce skewness and keep zero values valid.
df$log_dna <- log(df$dna_total_ng + 1)


############################################################
# 17. Handle missing values
############################################################

# Inspect missing values.
colSums(is.na(df))

# Add "Missing" level to cytology_group and replace NA values.
df$cytology_group <- as.factor(df$cytology_group)
levels(df$cytology_group) <- c(levels(df$cytology_group), "Missing")
df$cytology_group[is.na(df$cytology_group)] <- "Missing"

# If sample_year contains missing values, inspect them.
# The original analysis replaced one missing sample_year with 2016.
if ("sample_year" %in% names(df)) {
  df[is.na(df$sample_year), ]
  df$sample_year[is.na(df$sample_year)] <- 2016
}

# Check missing AGE or volume.
# Some models exclude samples with missing AGE or volume.
if ("volume" %in% names(df)) {
  table(df$qc_fail[is.na(df$AGE) | is.na(df$volume)])
  sum(is.na(df$AGE) | is.na(df$volume))

  # Create complete-case dataset excluding samples with missing AGE or volume.
  df_noNA <- df[!(is.na(df$AGE) | is.na(df$volume)), ]
} else {
  # If volume variable is not present, exclude only samples with missing AGE.
  table(df$qc_fail[is.na(df$AGE)])
  sum(is.na(df$AGE))

  df_noNA <- df[!is.na(df$AGE), ]
}


############################################################
# 18. Prepare variables for modelling
############################################################

# Convert variables to factor or numeric as required.
df_noNA$collector_type <- as.factor(df_noNA$collector_type)
df_noNA$cytology_group <- as.factor(df_noNA$cytology_group)
df_noNA$hpv_final <- as.factor(df_noNA$hpv_final)

if ("volume" %in% names(df_noNA)) {
  df_noNA$volume <- as.factor(df_noNA$volume)
}

df_noNA$log_dna <- as.numeric(df_noNA$log_dna)
df_noNA$AGE <- as.numeric(df_noNA$AGE)

if ("sample_year" %in% names(df_noNA)) {
  df_noNA$sample_year <- as.numeric(df_noNA$sample_year)
}

df_noNA$qc_fail <- as.factor(df_noNA$qc_fail)

# Inspect final modelling dataset.
str(df_noNA)


############################################################
# 19. Logistic regression models for QC failure
############################################################

# Short logistic model:
# Outcome: qc_fail
# Predictors: log-transformed DNA, collector type, cytology group,
# and cohort/case status.
modelround1_nocomplete <- glm(
  qc_fail ~ log_dna + collector_type + cytology_group + cohort_binary,
  data = df_noNA,
  family = binomial
)

summary(modelround1_nocomplete)

# Extended logistic model with additional variables.
# This model includes HPV, age and sample year.
if ("sample_year" %in% names(df_noNA)) {
  modelround1_more_variables <- glm(
    qc_fail ~ log_dna + collector_type + cytology_group +
      hpv_final + AGE + sample_year,
    data = df_noNA,
    family = binomial
  )
} else {
  modelround1_more_variables <- glm(
    qc_fail ~ log_dna + collector_type + cytology_group +
      hpv_final + AGE + year,
    data = df_noNA,
    family = binomial
  )
}

summary(modelround1_more_variables)


############################################################
# 20. Create OR table for short logistic model
############################################################

# Extract coefficients.
coef_table <- summary(modelround1_nocomplete)$coefficients

# Build OR table.
or_table_nocomplete <- data.frame(
  Variable = rownames(coef_table),
  Estimate = coef_table[, "Estimate"],
  Std_Error = coef_table[, "Std. Error"],
  p_value = coef_table[, "Pr(>|z|)"]
)

# Calculate OR and Wald 95% CI.
or_table_nocomplete <- or_table_nocomplete %>%
  mutate(
    OR = exp(Estimate),
    CI_lower = exp(Estimate - 1.96 * Std_Error),
    CI_upper = exp(Estimate + 1.96 * Std_Error)
  ) %>%
  select(Variable, OR, CI_lower, CI_upper, p_value) %>%
  mutate(
    OR = round(OR, 3),
    CI_lower = round(CI_lower, 3),
    CI_upper = round(CI_upper, 3),
    p_value = round(p_value, 4)
  )

# View and export OR table.
or_table_nocomplete
write_xlsx(or_table_nocomplete, "modelround1_nocomplete_OR_table.xlsx")


############################################################
# 21. Weighted logistic regression for QC failure
############################################################

# Because QC failures are rare, a weighted logistic model was tested.
# QC fail samples receive a larger weight to compensate for class imbalance.
model_weighted <- glm(
  qc_fail ~ log_dna + collector_type + cytology_group,
  data = df_noNA,
  family = binomial,
  weights = ifelse(df_noNA$qc_fail == 1, 20, 1)
)

summary(model_weighted)

# Extract weighted model coefficients.
coef_table_weighted <- summary(model_weighted)$coefficients

# Build OR table for weighted model.
or_table_weighted <- data.frame(
  Variable = rownames(coef_table_weighted),
  Estimate = coef_table_weighted[, "Estimate"],
  Std_Error = coef_table_weighted[, "Std. Error"],
  p_value = coef_table_weighted[, "Pr(>|z|)"]
)

or_table_weighted <- or_table_weighted %>%
  mutate(
    OR = exp(Estimate),
    CI_lower = exp(Estimate - 1.96 * Std_Error),
    CI_upper = exp(Estimate + 1.96 * Std_Error)
  ) %>%
  select(Variable, OR, CI_lower, CI_upper, p_value)

# View and export weighted OR table.
or_table_weighted
write_xlsx(or_table_weighted, "modelfinalglmOR_table.xlsx")
